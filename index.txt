<script>
  const windowEl = document.getElementById("timelineWindow");
  const track = document.getElementById("timelineTrack");
  const original = document.getElementById("timelineOriginal");
  const pauseButton = document.getElementById("pauseButton");
  const speedButton = document.getElementById("speedButton");

  // Duplicate the milestones so the animation loops seamlessly.
  const duplicate = original.cloneNode(true);
  duplicate.removeAttribute("id");
  duplicate.setAttribute("aria-hidden", "true");

  // Avoid duplicate IDs inside the cloned content.
  duplicate.querySelectorAll("[id]").forEach(el => el.removeAttribute("id"));
  track.appendChild(duplicate);

  const reducedMotion = window.matchMedia(
    "(prefers-reduced-motion: reduce)"
  );

  let paused = reducedMotion.matches;
  let fast = false;

  function updateAnimation() {
    track.style.animationDuration = fast ? "35s" : "65s";
    track.style.animationPlayState = paused ? "paused" : "running";
    pauseButton.textContent = paused
      ? "▶ Play animation"
      : "Ⅱ Pause animation";
    pauseButton.setAttribute("aria-pressed", String(paused));
    speedButton.textContent = fast ? "Speed: Fast" : "Speed: Normal";
    speedButton.setAttribute("aria-pressed", String(fast));
  }

  pauseButton.addEventListener("click", () => {
    paused = !paused;
    updateAnimation();
  });

  speedButton.addEventListener("click", () => {
    fast = !fast;
    updateAnimation();
  });

  // Pause when the visitor focuses the timeline with a keyboard.
  windowEl.addEventListener("focusin", () => {
    track.style.animationPlayState = "paused";
  });

  windowEl.addEventListener("focusout", () => {
    updateAnimation();
  });

  reducedMotion.addEventListener("change", event => {
    if (event.matches) paused = true;
    updateAnimation();
  });

  updateAnimation();
</script>
